\subsection*{Case 1: 2D Tensor Covariate}
\paragraph*{DGP.}
The tensor size is set to be $D=2$, $p_1=p_2=60$, and split by $p_1'=p_2'=12$ such that by $S^{(1)}=S^{(2)}=5$. Set the rank $R=4$, and the $s$-th block tensor for $j$-th eye of $i$-th patient is generated by:
\begin{align*}
    \mathcal{X}_{ij}^{(s)}=\sum_{r=1}^{R}x^*_{ijrs}\mathbf{x}_{r}^{(s1)}\circ\mathbf{x}_{r}^{(s2)},
    \qwhereq
    x_{ijrs}^*\iid \normal(0,1).
\end{align*}
and any $\mathbf{x}_{r}^{(sd)}$ and $\mathbf{x}_{r'}^{(sd)}$ are orthonormal and obtained from the QR decomposition of a random matrix. The coefficient function is set to be $\mathcal{A}^{(s)}(t)=\sum_{r=1}^{R} \alpha_{rs}(t)\mathbf{x}_r^{(s1)}\circ \mathbf{x}_r^{(s2)}$ where $\alpha_{rs}(t)$ has 4 choices:
\begin{align*}
    % \label{eq-sim-coef1}
    &\alpha_{1rs}(t)=
    5\sqrt{\frac{rs}{RS}}(t-0.2)^2,\quad
    \alpha_{2rs}(t)=
    \sqrt{\frac{rs}{RS}}
    \left(e^{-(3t-1)^2}-0.75\right),\quad
    \alpha_{3rs}(t)=
    \sqrt{\frac{rs}{RS}}\sin\bigl(2\pi(t-0.5)\bigr),\\
    &\alpha_{4rs}(t)=
    \sqrt{\frac{rs}{RS}}
    \Bigl[
    0.45\cdot 5(t-0.2)^2
    +0.35\bigl(e^{-(3t-1)^2}-0.75\bigr)
    +0.20\sin\bigl(2\pi(t-0.5)\bigr)
    \Bigr],\\
    &\alpha_{5rs}(t)=
    \sqrt{\frac{rs}{RS}}
    \Biggl[
    1.10\exp\!\left\{-\frac{(t-0.30)^2}{2(0.08)^2}\right\}
    -0.95\exp\!\left\{-\frac{(t-0.72)^2}{2(0.11)^2}\right\}
    \Biggr],\\
    &\alpha_{6rs}(t)=
    \sqrt{\frac{rs}{RS}}
    \Bigl[
    18(t-0.2)(t-0.55)(t-0.85)-0.05
    \Bigr]
    % \label{eq-sim-coef2}
\end{align*}
The one-way vector $\mathbf{Z}\in\bbR^{n\times p_0}$ is Gaussian ensemble matrix (iid entries) and $\bb=(2.0,1.0,-1.0,0.5)^\top$. We will vary $\sigma^2(t)$ to 4 subcases as Case 1(a)-(d).
\begin{align}
    \sigma^2_0(t)\equiv 1,\quad
    \sigma^2_1(t)=1+0.3\sin(2\pi t),\quad
    \sigma^2_2(t)=1+0.25\cos(2\pi t)+0.1 \sin(4\pi t),\qandq
    \sigma^2_3(t)=0.5+0.5\sin^2(\pi t).
\end{align}

\paragraph{Estimation Steps.} (1) We treat the paired data as iid, and get initial estimate $\hat{\mathbf{A}}^\dagger(t)$ and $\hat{\bb}^\dagger$. (2) Then estimate covariance structure $\hat{\sigma}^2(t)$ by eq(15)-(16) in AOAS draft.
% \begin{align*}
% \hat{\sigma}^{2}=
% \frac{\sum_i\sum_j
% \left[
% y_{ij}
% -
% \left\langle \mathbf{X}_{ij}^{*}, \hat{\mathbf{A}}^{\dagger}(t_i) \right\rangle
% -
% \mathbf{z}_{i}^{\top}\hat{\boldsymbol{\beta}}^{\dagger}
% \right]^{2}}{2n}
% ,\;
% \hat{\rho}=\frac{
% \sum_{i=1}^{n}
% \left[
% y_{i1}
% -
% \left\langle \mathbf{X}_{i1}^{*}, \hat{\mathbf{A}}^{\dagger}(t_i) \right\rangle
% -
% \mathbf{z}_{i}^{\top}\hat{\boldsymbol{\beta}}^{\dagger}
% \right]
% \left[
% y_{i2}
% -
% \left\langle \mathbf{X}_{i2}^{*}, \hat{\mathbf{A}}^{\dagger}(t_i) \right\rangle
% -
% \mathbf{z}_{i}^{\top}\hat{\boldsymbol{\beta}}^{\dagger}
% \right]
% }{n\hat{\sigma}^2}.
% \end{align*}
(3) Get final estimates $\hat{\mathbf{A}}^*(t)$ and $\hat{\bb}^*$ from eq(14)-(15) of AOAS draft.

\paragraph{Simulation Results.} We vary $n=1000/2000$, coefficients $\alpha_{1rs}(t),\dots,\alpha_{4rs}(t)$, and correlation $\rho=0/0.3/0.6/0.9$. We report the error of both initial estimates $(\hat{\mathbf{A}}^\dagger,\;\hat{\bb}^\dagger)$ and final estimates $(\hat{\mathbf{A}}^*,\;\hat{\bb}^*)$, also the error of $(\hat{\sigma}^2, \hat{\rho})$.
