\begin{center}
    \large\textsc{Simulation Ideas}
\end{center}

For each observation $i\in [n]$, denote by its tensor covariate $\mathcal{X}_i\in\bbR^{p_1\times\cdots\times p_D}$, one-way covariates $\mathbf{z}_i\in\bbR^{p_0}$, and two responses $y_{ij}\in\bbR$ where $j=1,2$ represents two eyes. Let $t\in[0,1]$ be the observation time, $\mathcal{A}(t)=(\alpha_{j_1,\dots,j_d}(t))\in\bbR^{p_1\times\cdots\times p_D}$ be the unknown tensor coefficient function, and $\bb\in\bbR^{p_0}$ be the coefficients. The model is assumed to be
\begin{align}
\label{eq-model0}
y_{ij}
&= \left\langle \mathcal{X}_{ij}, \mathcal{A}(t_i) \right\rangle
   + \mathbf{z}_i^\top \boldsymbol{\beta}
   + \epsilon_{ij} = \sum_{s=1}^S
   \left\langle \mathcal{X}_{ij}^{(s)}, \mathcal{A}^{(s)}(t_i) \right\rangle
   + \mathbf{z}_i^\top \boldsymbol{\beta}
   + \epsilon_{ij} \\
\intertext{by a rank-$R$ CP decomposition, we can reorganize it in a matrix form}
&\approx \sum_{s=1}^S \sum_{r=1}^R
   x_{ijrs}^{*}
   \left\langle
   \mathcal{U}_r^{(s)},
   \mathcal{A}^{(s)}(t_i)
   \right\rangle
   + \mathbf{z}_i^\top \boldsymbol{\beta}
   + \epsilon_{ij} 
   = \left\langle \mathbf{X}_{ij}^{*}, \mathbf{A}(t_i) \right\rangle
   + \mathbf{z}_i^\top \boldsymbol{\beta}
   + \epsilon_{ij},
   \nonumber
\end{align}
where matrix $\mathbf{X}_{ij}^{*}\coloneq(x_{ijrs}^*)\in\bbR^{R\times S}$, and $\mathbf{A}(t)\coloneq (\alpha_{rs}(t))\in\bbR^{R\times S}$ with $\alpha_{rs}(t)\coloneq \innerprod{\mathcal{U}_r^{(s)}}{\mathcal{A}^{(s)}(t)}$. Within one patient, the eyes are correlated by
\begin{align*}
    \begin{bmatrix}
        \epsilon_{i1}\\ \epsilon_{i2}
    \end{bmatrix}\sim 
    \mvnorm\begin{pmatrix}
        \mathbf{0},\begin{bmatrix}
            \sigma^2 & \rho\sigma^2\\ 
            \rho\sigma^2 & \sigma^2
        \end{bmatrix}
    \end{pmatrix}.
\end{align*}

\subsection{Paired Simulation for Case I}
\paragraph*{DGP.}
The tensor size is set to be $D=2$, $p_1=p_2=80$, and $S=16$. We split the tensor by $S^{(1)}=S^{(2)}=4$ and $p_1'=p_2'=20$. Set the rank $R=10$, and the $s$-th block tensor for $j$-th eye of $i$-th patient is generated by:
\begin{align*}
    \mathcal{X}_{ij}^{(s)}=\sum_{r=1}^{R}x^*_{ijrs}\mathbf{x}_{r}^{(s1)}\circ\mathbf{x}_{r}^{(s2)},
    \qwhereq
    x_{ijrs}^*\iid \normal(0,1)\qquad
    \text{\textcolor{red}{\textbf{Remark.} Can consider correlation between $j=1,2$.}}
\end{align*}
and any $\mathbf{x}_{r}^{(sd)}$ and $\mathbf{x}_{r'}^{(sd)}$ are orthonormal and obtained from the QR decomposition of a random matrix. The coefficient function is set to be $\mathcal{A}^{(s)}(t)=\sum_{r=1}^{R} \alpha_{rs}(t)\mathbf{x}_r^{(s1)}\circ \mathbf{x}_r^{(s2)}$ where $\alpha_{rs}(t)$ has 4 choises:
\begin{align}
    \label{eq-sim-coef1}
    \alpha_{rs}(t)=4\sqrt{\frac{rs}{RS}}\times (t-0.5)^2 
    \to \mathcal{A}_1(t) &\qandq
    \mathcal{A}_2(t) \leftarrow \alpha_{rs}(t)=\sqrt{\frac{rs}{RS}}\times t^{0.5}, \\ 
    \alpha_{rs}(t)=1.75
    \sqrt{\frac{rs}{RS}}
    \left( e^{-(3t-1)^2}+e^{-(4t-3)^2}-0.75\right)
    \to \mathcal{A}_3(t)
    &\qandq
    \mathcal{A}_4(t) \leftarrow
    \alpha_{rs}(t)=\sqrt{\frac{rs}{RS}}\sin\bigl(2\pi(t-0.5)\bigr). 
    \label{eq-sim-coef2}
\end{align}
The one-way vector $\mathbf{Z}\in\bbR^{n\times p_0}$ is Gaussian ensemble matrix (iid entries) and $\bb=(3,3)^\top$. Finally the responses are generated by model eq(\ref{eq-model0}).

\paragraph{Estimation Steps.} (1) We treat the paired data as iid, and get initial estimate $\hat{\mathbf{A}}^\dagger(t)$ and $\hat{\bb}^\dagger$. (2) Then estimate
\begin{align*}
\hat{\sigma}^{2}=
\frac{\sum_i\sum_j
\left[
y_{ij}
-
\left\langle \mathbf{X}_{ij}^{*}, \hat{\mathbf{A}}^{\dagger}(t_i) \right\rangle
-
\mathbf{z}_{i}^{\top}\hat{\boldsymbol{\beta}}^{\dagger}
\right]^{2}}{2n}
,\;
\hat{\rho}=\frac{
\sum_{i=1}^{n}
\left[
y_{i1}
-
\left\langle \mathbf{X}_{i1}^{*}, \hat{\mathbf{A}}^{\dagger}(t_i) \right\rangle
-
\mathbf{z}_{i}^{\top}\hat{\boldsymbol{\beta}}^{\dagger}
\right]
\left[
y_{i2}
-
\left\langle \mathbf{X}_{i2}^{*}, \hat{\mathbf{A}}^{\dagger}(t_i) \right\rangle
-
\mathbf{z}_{i}^{\top}\hat{\boldsymbol{\beta}}^{\dagger}
\right]
}{n\hat{\sigma}^2}.
\end{align*}
(3) Get final estimates $\hat{\mathbf{A}}^*(t)$ and $\hat{\bb}^*$ from Eq(14) and Eq(15) of AOAS draft.

\paragraph{Simulation Results.} We vary $n=500/1000/2000$, coefficients $\mathcal{A}_1(t),\dots,\mathcal{A}_4(t)$, and correlation $\rho=0/0.3/0.6/0.9$. We report the error of both initial estimates $(\hat{\mathbf{A}}^\dagger,\;\hat{\bb}^\dagger)$ and final estimates $(\hat{\mathbf{A}}^*,\;\hat{\bb}^*)$, also the error of $(\hat{\sigma}^2, \hat{\rho})$.
